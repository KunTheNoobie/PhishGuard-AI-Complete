## Appendix B: FastAPI Asynchronous Route Implementation Snippet

The following Python codebase illustrates the non-blocking execution of the primary `POST /api/v1/analyze/semantics` endpoint, highlighting the integration of `asyncio.to_thread()` tensor offloading, `asyncio.gather()` parallel coroutines, and `aiosqlite` connection pooling.

```python
"""
PhishGuard-AI: Asynchronous Multi-Modal Threat Orchestrator
Author: Liew Yi Ler (Student ID: 25WMR09747)
Module: Semantic Threat Intelligence and Mule Account Verification Engine
"""

from fastapi import APIRouter, Depends, HTTPException, Request, Security
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from pydantic import BaseModel, HttpUrl
from typing import Any, Dict, List, Optional
import asyncio
import time

router = APIRouter(prefix="/api/v1", tags=["Threat Analysis"])
security = HTTPBearer()

class SemanticAnalysisRequest(BaseModel):
    url: str
    text_content: str
    origin: Optional[str] = None
    extract_mules: bool = True

@router.post("/analyze/semantics")
async def analyze_semantics_endpoint(
    request: Request,
    payload: SemanticAnalysisRequest,
    credentials: HTTPAuthorizationCredentials = Security(security)
) -> Dict[str, Any]:
    # 1. Enforce Cryptographic Bearer Token Authentication
    if credentials.credentials != getattr(request.app.state, "secret_token", "phishguard_secret_key_2026"):
        raise HTTPException(status_code=401, detail="Unauthorized: Invalid Bearer Token")
        
    start_time = time.perf_counter()
    
    # 2. Fast-Path In-Memory Whitelist Lookup (0ms overhead)
    semantic_engine = request.app.state.semantic_engine
    if semantic_engine.is_whitelisted(payload.url):
        return {
            "status": "success",
            "verdict": "SAFE",
            "risk_score": 0.00,
            "bert_score": 0.00,
            "is_whitelisted": True,
            "brand_impersonation": {"is_impersonation": False, "targeted_brand": None, "similarity_index": 0.0},
            "mule_accounts_flagged": [],
            "execution_time_ms": round((time.perf_counter() - start_time) * 1000, 2),
            "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S+08:00")
        }
        
    # 3. Asynchronous Parallel Execution via asyncio.gather()
    # Task A: Offload CPU-bound PyTorch BERT forward pass to worker thread
    bert_task = asyncio.to_thread(
        semantic_engine.predict_threat_probability,
        payload.text_content,
        payload.url
    )
    
    # Task B: Execute non-blocking C-bytecode Regex & SQLite mule registry query
    mule_scanner = request.app.state.mule_scanner
    mule_task = mule_scanner.scan_and_verify_async(payload.text_content)
    
    # Run AI inference and database lookup concurrently
    bert_score, mule_matches = await asyncio.gather(bert_task, mule_task)
    
    # 4. Multi-Vector Risk Calibration & Decision Aggregation
    is_mule = len(mule_matches) > 0
    is_threat = (bert_score >= 0.70) or is_mule
    final_verdict = "BLOCK_RENDER" if is_threat else "SAFE"
    calibrated_risk = max(bert_score, 0.985 if is_mule else 0.0)
    
    # 5. Non-Blocking Telemetry Persistence & SSE Push
    db = request.app.state.db
    asyncio.create_task(db.log_telemetry(payload.url, bert_score, final_verdict))
    
    elapsed_ms = round((time.perf_counter() - start_time) * 1000, 2)
    
    return {
        "status": "success",
        "verdict": final_verdict,
        "risk_score": calibrated_risk,
        "bert_score": bert_score,
        "is_whitelisted": False,
        "brand_impersonation": semantic_engine.evaluate_brand_impersonation(payload.url),
        "mule_accounts_flagged": mule_matches,
        "execution_time_ms": elapsed_ms,
        "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S+08:00")
    }
```
