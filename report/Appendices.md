## Appendix B: FastAPI Asynchronous Route Implementation Snippet

The following Python codebase demonstrates the production asynchronous endpoint implementation utilizing the FastAPI framework. This code illustrates the practical application of ASGI non-blocking concurrency principles discussed in Chapters 4 and 5, highlighting Pydantic v2 validation models, `asyncio.to_thread()` tensor offloading, and `asyncio.gather()` parallelization.

```python
"""
PhishGuard-AI: Asynchronous Multi-Modal Threat Orchestrator
Author: Liew Yi Ler (Student ID: 25WMR09747)
Module: Semantic Threat Intelligence and Mule Account Verification Engine
"""

from fastapi import APIRouter, Depends, HTTPException, Request, Security, BackgroundTasks
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from pydantic import BaseModel, HttpUrl
from typing import Any, Dict, List, Optional
import asyncio
import time
import logging

router = APIRouter(prefix="/api/v1", tags=["Threat Analysis"])
security = HTTPBearer()
logger = logging.getLogger(__name__)

# Pydantic v2 Schema for strict JSON payload validation
class WebPayload(BaseModel):
    url: str
    text_content: str
    origin: Optional[str] = None
    extract_mules: bool = True

@router.post("/analyze/semantics", status_code=200)
async def analyze_web_payload(
    request: Request,
    payload: WebPayload,
    background_tasks: BackgroundTasks,
    credentials: HTTPAuthorizationCredentials = Security(security)
) -> Dict[str, Any]:
    try:
        # Step 1: Enforce Cryptographic Bearer Token Authentication
        expected_token = getattr(request.app.state, "secret_token", "phishguard_secret_key_2026")
        if credentials.credentials != expected_token:
            raise HTTPException(status_code=401, detail="Unauthorized: Invalid Bearer Token")
            
        start_time = time.perf_counter()
        
        # Step 2: Fast-Path In-Memory Whitelist Lookup (< 1ms overhead)
        semantic_engine = request.app.state.semantic_engine
        if semantic_engine.is_whitelisted(payload.url):
            return {
                "status": "success",
                "verdict": "SAFE",
                "risk_score": 0.00,
                "bert_score": 0.00,
                "is_whitelisted": True,
                "brand_impersonation": {"is_impersonation": False, "targeted_brand": None, "similarity_index": 0.0},
                "mule_accounts_flagged": [],
                "execution_time_ms": round((time.perf_counter() - start_time) * 1000, 2),
                "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S+08:00")
            }
            
        # Step 3: Concurrent Parallel Execution via asyncio.gather()
        # Task A: Offload CPU-bound PyTorch BERT forward pass to worker thread
        bert_task = asyncio.to_thread(
            semantic_engine.predict_threat_probability,
            payload.text_content,
            payload.url
        )
        
        # Task B: Execute non-blocking C-bytecode Regex & SQLite mule registry query
        mule_scanner = request.app.state.mule_scanner
        mule_task = mule_scanner.scan_and_verify_async(payload.text_content)
        
        # Await completion of both parallel tasks simultaneously
        bert_score, mule_matches = await asyncio.gather(bert_task, mule_task)
        
        # Step 4: Multi-Vector Risk Calibration & Decision Aggregation
        is_mule = len(mule_matches) > 0
        is_threat = (bert_score >= 0.70) or is_mule
        final_verdict = "BLOCK_RENDER" if is_threat else "SAFE"
        calibrated_risk = max(bert_score, 0.985 if is_mule else 0.0)
        
        # Step 5: Non-Blocking Telemetry Persistence (Background Task)
        db = request.app.state.db
        background_tasks.add_task(db.log_telemetry, payload.url, bert_score, final_verdict)
        
        elapsed_ms = round((time.perf_counter() - start_time) * 1000, 2)
        
        return {
            "status": "success",
            "verdict": final_verdict,
            "risk_score": calibrated_risk,
            "bert_score": bert_score,
            "is_whitelisted": False,
            "brand_impersonation": semantic_engine.evaluate_brand_impersonation(payload.url),
            "mule_accounts_flagged": mule_matches,
            "execution_time_ms": elapsed_ms,
            "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S+08:00")
        }
        
    except ValueError as val_err:
        logger.error(f"Validation Error: {val_err}")
        raise HTTPException(status_code=422, detail=f"Unprocessable Entity: {str(val_err)}")
    except Exception as e:
        logger.critical(f"Inference Engine Failure: {e}")
        raise HTTPException(status_code=500, detail="Internal Server Error: AI inference pipeline failed.")
```
